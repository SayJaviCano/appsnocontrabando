# Sustituye el bloque anterior completo.

set $seourl 1;
if ($uri = /) {
    set $seourl 0;
}
if ($uri ~* \.(css|js|jpg|jpeg|png|gif|ico|svg|woff|woff2|ttf|eot|otf|webp|map|txt|xml|json)$) {
    set $seourl 0;
}
if (-f $request_filename) {
    set $seourl 0;
}
if ($seourl = 1) {
    rewrite ^/(.*)$ /index.php?__path=$1 last;
}
